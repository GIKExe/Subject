# Группировка + Нормализация
# Группировка: уникальные ID, обратные индексы и размеры групп
unique_filt, inverse, counts = np.unique(data['filter_id'], return_inverse=True, return_counts=True)
n_groups = len(unique_filt)

print(f'Количество групп: {n_groups}')
# Конвертация в нативные Python-типы для чистого вывода без указания dtype
print('Мощность групп:', dict(zip(unique_filt.tolist(), counts.tolist())))

# Векторное вычисление среднего и стандартного отклонения по группам
turb = data['turb'].astype(np.float64)

group_sum = np.bincount(inverse, weights=turb)
group_sq_sum = np.bincount(inverse, weights=turb**2)

group_means = group_sum / counts
group_vars = group_sq_sum / counts - group_means**2
group_vars = np.maximum(group_vars, 0.0)
group_stds = np.sqrt(group_vars)

# Разворачивание групповых статистик на каждую строку массива
means_per_row = group_means[inverse]
stds_per_row = group_stds[inverse]

# Вычисление Z-score
turb_z = ((turb - means_per_row) / (stds_per_row + 1e-8)).astype(np.float32)

# Расширение dtype и векторное копирование данных
dtype_z = np.dtype(data.dtype.descr + [('turb_z', np.float32)])
data_z = np.empty(data.shape, dtype=dtype_z)

# Копирование исходных полей в новый массив
data_z[list(data.dtype.names)] = data
data_z['turb_z'] = turb_z

print('Z-score нормализация turb по группам выполнена.')
np.save('/content/data_step3.npy', data_z)
